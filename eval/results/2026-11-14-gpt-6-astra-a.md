### 4. Phone checkout flow
```wirelith
wirelith 0.1

screen "Cart" phone #cart
  column pad=md gap=md
    h1 "Your cart"
    text "Step 1 of 3" muted
    card "Everyday backpack"
      text "Quantity: 1"
      text "$64.00"
    divider
    row justify=between
      text "Shipping"
      text "Free"
    row justify=between
      h3 "Total"
      h3 "$64.00"
    button "Continue to shipping" primary width=fill -> #shipping

screen "Shipping address" phone #shipping
  column pad=md gap=md
    header
      button "Back to cart" icon=arrow-left -> #cart
    h1 "Shipping address"
    text "Step 2 of 3" muted
    input "Full name" value="Alex Morgan" required
    input "Street address" value="18 Willow Road" required
    input "City" value="London" required
    input "Postcode" value="SW1A 1AA" required
    select "Country" [United Kingdom, Ireland, France] value="United Kingdom"
    text "Total: $64.00"
    button "Place order" primary width=fill -> #confirmed

screen "Shipping address" phone #shipping-error of=#shipping state="Invalid postcode"
  column pad=md gap=md
    header
      button "Back to cart" icon=arrow-left -> #cart
    h1 "Shipping address"
    text "Step 2 of 3" muted
    error "Check your postcode before placing your order."
    input "Full name" value="Alex Morgan" required
    input "Street address" value="18 Willow Road" required
    input "City" value="London" required
    input "Postcode" value="123" required error="Enter a valid UK postcode, such as SW1A 1AA."
    select "Country" [United Kingdom, Ireland, France] value="United Kingdom"
    text "Total: $64.00"
    button "Place order" primary width=fill -> #confirmed

screen "Order confirmed" phone #confirmed
  column pad=lg gap=md align=center
    icon "check"
    h1 "Order confirmed"
    text "Step 3 of 3" muted
    text "Thank you, Alex. Your order is on its way."
    badge "Order WL-1042" success
    text "Everyday backpack · Quantity 1"
    text "Total paid: $64.00"
    button "Back to cart" width=fill -> #cart

flow #shipping -> #shipping-error "Place order with an invalid postcode"
flow #shipping-error -> #confirmed "Correct the postcode and place order"
```
